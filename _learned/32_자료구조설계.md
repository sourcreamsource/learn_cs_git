## 🟢 1. 전체 구조 한눈에 보기

## 🟢 2. Commit 클래스 설계  

### 🟡 필드 설명  
| 필드 | 타입 | 설명 | 예시 |  
|------|------|------|------|
| `hash` | `str` | 고유 식별자 | `"a1b2c3"` |  
| `message` | `str` | 커밋 설명 | `"로그인 기능 추가"` |  
| `author` | `str` | 작성자 이름 | `"홍길동"` |  
| `timestamp` | `str` | 생성 시각 | `"2024-01-15T09:30:00"` |  
| `parents` | `List[str]` | 부모 해시들 | `["parent_hash"]` 또는 `[]` (최초) |  

### 🟡 파이썬 구현 (dataclass 추천 - Python 3.7+)  
```python
from dataclasses import dataclass, field  
from typing import List  
from datetime import datetime  

@dataclass  
class Commit:  
    """커밋 하나 = 스냅샷 메타데이터"""  
    hash: str  
    message: str  
    author: str  
    timestamp: str  
    parents: List[str] = field(default_factory=list)  
    
    def __post_init__(self):  
        # 부모 리스트가 None이면 빈 리스트로  
        if self.parents is None:  
            self.parents = []  
    
    def is_root(self) -> bool:  
        """최초 커밋(루트)인지 확인"""  
        return len(self.parents) == 0  
    
    def is_merge(self) -> bool:  
        """Merge 커밋인지 확인 (부모 2개 이상)"""  
        return len(self.parents) >= 2  
    
    def __repr__(self):  
        short_hash = self.hash[:7]  
        return f"Commit({short_hash}, '{self.message[:20]}', by {self.author})"  
```

### 🟡 일반 클래스로도 가능 (dataclass 안 쓸 때)  
```python
class Commit:  
    def __init__(self, hash_val, message, author, timestamp, parents=None):  
        self.hash = hash_val  
        self.message = message  
        self.author = author  
        self.timestamp = timestamp  
        self.parents = parents or []  # None 방지  
    
    def is_root(self):  
        return len(self.parents) == 0  
    
    def is_merge(self):  
        return len(self.parents) >= 2  
```

## 🟢 3. Repository 클래스 설계  

### 🟡 책임  
- **커밋 저장/조회** (해시맵)  
- **브랜치 관리** (이름 ↔ 해시)  
- **HEAD 관리** (현재 위치)  
- **역색인 관리** (검색용)  
- **명령어 실행 진입점**  

### 🟡 파이썬 구현  
```python
from collections import deque  
from typing import Dict, List, Optional, Set  
import hashlib  
import time  
import random  

class Repository:  
    def __init__(self):  
        # 핵심 저장소  
        self.commits: Dict[str, Commit] = {}      # hash -> Commit  
        self.branches: Dict[str, str] = {}        # branch_name -> commit_hash  
        self.head: Optional[str] = None           # 현재 브랜치명  
        self.current_user: Optional[str] = None   # 현재 작성자  
        
        # 역색인 (검색용)  
        self.keyword_index: Dict[str, List[str]] = {}  # word -> [hashes]  
        self.author_index: Dict[str, List[str]] = {}   # author -> [hashes]  
        
        # 해시 중복 방지용  
        self._used_hashes: Set[str] = set()  
    
    # ========== 해시 생성 ==========  
    def _generate_hash(self) -> str:  
        """고유 해시 생성 (중복 방지)"""  
        while True:  
            # 타임스탬프 + 랜덤 + 카운터 조합  
            data = f"{time.time()}:{random.random()}:{len(self._used_hashes)}"  
            h = hashlib.sha256(data.encode()).hexdigest()[:12]  
            if h not in self._used_hashes:  
                self._used_hashes.add(h)  
                return h  
    
    # ========== 커밋 관리 ==========  
    def add_commit(self, message: str) -> Commit:  
        """새 커밋 생성 및 저장"""  
        # 부모: 현재 HEAD가 가리키는 커밋  
        parent_hash = self.branches.get(self.head) if self.head else None  
        parents = [parent_hash] if parent_hash else []  
        
        # 커밋 객체 생성  
        commit = Commit(  
            hash=self._generate_hash(),  
            message=message,  
            author=self.current_user,  
            timestamp=datetime.now().isoformat(timespec='seconds'),  
            parents=parents  
        )  
        
        # 저장소에 추가  
        self.commits[commit.hash] = commit  
        
        # 브랜치 포인터 이동 (현재 브랜치가 새 커밋 가리키게)  
        if self.head:  
            self.branches[self.head] = commit.hash  
        
        # 역색인 업데이트  
        self._update_index(commit)  
        
        return commit  
    
    def get_commit(self, hash_val: str) -> Optional[Commit]:  
        """커밋 조회 (O(1))"""  
        return self.commits.get(hash_val)  
    
    # ========== 역색인 업데이트 ==========  
    def _update_index(self, commit: Commit):  
        """커밋 추가 시 역색인 갱신"""  
        # 1. 작성자 색인  
        if commit.author not in self.author_index:  
            self.author_index[commit.author] = []  
        if commit.hash not in self.author_index[commit.author]:  
            self.author_index[commit.author].append(commit.hash)  
        
        # 2. 키워드 색인 (공백 분리 + 소문자)  
        words = commit.message.lower().split()  
        for word in words:  
            if word not in self.keyword_index:  
                self.keyword_index[word] = []  
            if commit.hash not in self.keyword_index[word]:  
                self.keyword_index[word].append(commit.hash)  
    
    # ========== 브랜치 관리 ==========  
    def create_branch(self, branch_name: str) -> bool:  
        """현재 HEAD 위치에 새 브랜치 생성"""  
        if not self.head:  
            return False  
        current_commit = self.branches[self.head]  
        self.branches[branch_name] = current_commit  
        return True  
    
    def switch_branch(self, branch_name: str) -> bool:  
        """브랜치 전환"""  
        if branch_name not in self.branches:  
            return False  
        self.head = branch_name  
        return True  
    
    def get_current_commit_hash(self) -> Optional[str]:  
        """현재 HEAD 커밋 해시 반환"""  
        if self.head:  
            return self.branches.get(self.head)  
        return None  
    
    # ========== 검색 ==========  
    def search_by_keyword(self, keyword: str) -> List[str]:  
        """키워드로 커밋 해시 리스트 찾기"""  
        return self.keyword_index.get(keyword.lower(), [])  
    
    def search_by_author(self, author: str) -> List[str]:  
        """작성자로 커밋 해시 리스트 찾기"""  
        return self.author_index.get(author, [])  
    
    # ========== 그래프 연산용 헬퍼 ==========  
    def get_children_graph(self) -> Dict[str, List[str]]:  
        """부모→자식 방향 그래프 생성 (위상 정렬용)"""  
        children = {h: [] for h in self.commits}  
        for hash_val, commit in self.commits.items():  
            for parent_hash in commit.parents:  
                if parent_hash in children:  
                    children[parent_hash].append(hash_val)  
        return children  
    
    def get_undirected_graph(self) -> Dict[str, List[str]]:  
        """무방향 그래프 생성 (최단 경로용)"""  
        graph = {h: [] for h in self.commits}  
        for hash_val, commit in self.commits.items():  
            for parent_hash in commit.parents:  
                graph[hash_val].append(parent_hash)  
                graph[parent_hash].append(hash_val)  
        return graph  
```

## 🟢 4. 역색인 (Index) 상세 구조  

### 🟡 🔥🔥🔥🔥🔥 키워드 색인  
```python
# 구조: {단어: [커밋해시들]}  
keyword_index = {  
    "로그인": ["a1b2c3", "d4e5f6"],  
    "기능": ["a1b2c3", "g7h8i9"],  
    "버튼": ["d4e5f6"],  
    "추가": ["a1b2c3", "g7h8i9", "j1k2l3"],  
}  

# 검색: O(1) 조회  
def search_keyword(index, word):  
    return index.get(word.lower(), [])  
```

```python
    "김철수": ["j1k2l3", "m4n5o6"],

def search_author(index, name):
```

## 🟢 5. 그래프 표현: 인접 리스트 (Adjacency List)  

### 🟡 왜 인접 리스트?  
| 표현 | 공간 | 조회 | 특징 |  
|------|------|------|------|
| **인접 행렬** | O(V²) | O(1) | 밀집 그래프용 |  
| **인접 리스트** | **O(V + E)** | O(차수) | **희소 그래프용 (Git은 이게 맞음!)** |  

> Git 커밋 그래프: 커밋 1개당 부모 평균 1~2개 → **매우 희소** → 인접 리스트 최적  

### 🟡 세 가지 그래프 변환  

```python
# 1. 원본: 자식 → 부모 (Commit.parents)

children_graph = {  
    "C0": ["C1"],

    "C2": ["C3"],  
    "C4": ["C5"],  
    "C3": ["C5"],

# 3. 무방향 (최단 경로 PATH용)
```

## 🟢 6. CLI 파서 (명령어 해석기)  

### 🟡 입력 파싱 로직  
```python
import shlex  # 따옴표 처리용 표준 라이브러리  

def parse_command(line: str):  
    """  
    입력: 'commit "Add login feature"'  
    출력: ('commit', ['Add login feature'])  
    """  
    line = line.strip()  
    if not line:  
        return None, []  
    
    # shlex.split: 따옴표 안 공백 보존하며 분리  
    parts = shlex.split(line, posix=True)  
    command = parts[0].upper()  # 대소문자 무시  
    args = parts[1:]  
    
    return command, args  

# 옵션 파싱 예: LOG --sort-by=date  
def parse_options(args):  
    options = {}  
    positional = []  
    
    for arg in args:  
        if arg.startswith('--'):  
            # --key=value 형태  
            if '=' in arg:  
                key, value = arg[2:].split('=', 1)  
                options[key] = value  
            else:  
                options[arg[2:]] = True  
        else:  
            positional.append(arg)  
    
    return options, positional  
```

## 🟢 7. 전체 흐름: 명령어 처리 파이프라인  

```
사용자 입력  
    │  
    ▼  
parse_command()  ──► command, args  
    │  
    ▼  
dispatcher[command](args)  ──► 해당 함수 실행  
    │  
    ├── INIT    ──► repo.init(user)  
    ├── BRANCH  ──► repo.create_branch(name)  
    ├── SWITCH  ──► repo.switch_branch(name)  
    ├── COMMIT  ──► repo.add_commit(message)  
    ├── LOG     ──► repo.log(sort_by=...)  
    ├── PATH    ──► repo.path(c1, c2)  
    ├── ANCESTORS ──► repo.ancestors(hash)  
    ├── SEARCH  ──► repo.search(keyword/author)  
    │  
    ▼  
결과 출력  
```

### 🟡 디스패처 구현  
```python
class MiniGitCLI:  
    def __init__(self):  
        self.repo = Repository()  
        self.commands = {  
            'INIT': self.cmd_init,  
            'BRANCH': self.cmd_branch,  
            'SWITCH': self.cmd_switch,  
            'COMMIT': self.cmd_commit,  
            'LOG': self.cmd_log,  
            'PATH': self.cmd_path,  
            'ANCESTORS': self.cmd_ancestors,  
            'SEARCH': self.cmd_search,  
            'EXIT': self.cmd_exit,  
            'QUIT': self.cmd_exit,  
            'HELP': self.cmd_help,  
        }  
    
    def run(self):  
        print("Mini Git 시작! (exit/quit로 종료)")  
        while True:  
            try:  
                line = input("mini-git> ").strip()  
                if not line:  
                    continue  
                
                cmd, args = parse_command(line)  
                cmd = cmd.upper()  
                
                if cmd in self.commands:  
                    self.commands[cmd](args)  
                else:  
                    print(f"알 수 없는 명령: {cmd}")  
                    
            except KeyboardInterrupt:  
                print("\n종료합니다.")  
                break  
            except EOFError:  
                break  
```

## 🟢 8. 미니 실습: 클래스 직접 만들어보기  

```python
# 1. Commit 생성 테스트  
from datetime import datetime  

c1 = Commit(  
    hash="a1b2c3",  
    message="첫 커밋",  
    author="홍길동",  
    timestamp=datetime.now().isoformat(),  
    parents=[]  
)  
print(c1)  
print("루트인가?", c1.is_root())  

c2 = Commit(  
    hash="d4e5f6",  
    message="두 번째",  
    author="홍길동",  
    timestamp=datetime.now().isoformat(),  
    parents=["a1b2c3"]  
)  
print(c2)  
print("부모:", c2.parents)  

# 2. Repository 테스트  
repo = Repository()  
repo.current_user = "홍길동"  
repo.head = "main"  
repo.branches["main"] = None  # 아직 커밋 없음  

# 첫 커밋  
commit1 = repo.add_commit("초기화")  
print(f"커밋1: {commit1.hash} - {commit1.message}")  

# 브랜치 생성  
repo.create_branch("feature")  
print(f"브랜치들: {repo.branches}")  

# 브랜치 전환  
repo.switch_branch("feature")  
print(f"현재 브랜치: {repo.head}")  

# 커밋 더 추가  
commit2 = repo.add_commit("기능 추가")  
print(f"커밋2: {commit2.hash} - 부모: {commit2.parents}")  

# 검색 테스트  
print(f"키워드 '기능' 검색: {repo.search_by_keyword('기능')}")  
print(f"작성자 '홍길동' 검색: {repo.search_by_author('홍길동')}")  

# 그래프 변환 테스트  
print(f"자식 그래프: {repo.get_children_graph()}")  
print(f"무방향 그래프: {repo.get_undirected_graph()}")  
```
